  if(EditorApplication.isPlaying||SessionState.GetBool(Key+"active",false))throw new Exception("Fresh edit state required");

  Directory.CreateDirectory(root);SessionState.SetString(Key+"root",root);SessionState.SetString(Key+"jobs",JsonConvert.SerializeObject(jobs));Index=0;
  SessionState.SetBool(Key+"active",true);State="enter";ready=EditorApplication.timeSinceStartup+1;Write("STARTED");return "Started "+jobs.Length+" jobs; no geometry rebase";
 }

 static void RestoreClock(){if(!clockHeld)return;Time.captureDeltaTime=oldCaptureDelta;clockHeld=false;}
 static void Helpers()
 {
  // Mouse isolation and held-key reinjection were removed after ablations.
  // Legacy suites now own their original input stimulus without this runner rewriting it.
  if(!Application.isPlaying){RestoreClock();return;}
  // A, opt-in timing control: legacy's 0.075UU decimation assertion assumes
  // more than 26.67 simulation frames/s at 2UU/s. Low Editor render FPS can violate
  // that assumption. A 60Hz capture clock separates that automation precondition
  // from geometry without changing any expected value or controller/sampler code.
  // Raw-clock ablations are kept as failures; this is NOT identical wall timing.
  if(Current.clock){if(!clockHeld){oldCaptureDelta=Time.captureDeltaTime;clockHeld=true;}Time.captureDeltaTime=1f/60f;}else RestoreClock();
  // A: focus/repaint schedules real PlayerLoop/input. Does not set product state.
  var game=EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView"));game.Focus();game.GetType().GetMethod("RepaintImmediately",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic).Invoke(game,null);EditorApplication.QueuePlayerLoopUpdate();
 }

 static void Tick()
 {
  if(!SessionState.GetBool(Key+"active",false)){TryRequest();return;}
  try{
   if(Index>=Jobs.Length){RestoreClock();SessionState.SetBool(Key+"active",false);State="complete";Write("COMPLETE");Capture(Root);return;}
   Helpers();Trace();if(EditorApplication.isCompiling||EditorApplication.isUpdating||EditorApplication.timeSinceStartup<ready)return;
   if(State=="original"){if(SessionState.GetBool("ConfidenceRegression.active",false))return;Index++;State="enter";Write("ORIGINAL_FINISHED");return;}
   if(State=="enter"){
    if(Current.suite=="Original103"){State="original";typeof(ConfidenceRegressionRunner).GetMethod("Start",BindingFlags.NonPublic|BindingFlags.Static).Invoke(null,new object[]{Folder});Write("ORIGINAL_STARTED");return;}
    if(Current.suite=="Confidence"){ShipyardConfidenceChecks.Run(Folder);Index++;Write("CONFIDENCE_FINISHED");return;}
    State="changing";Write("ENTERING");EditorApplication.isPlaying=true;
   }else if(State=="ready"){BeginSuite();State="running";ready=EditorApplication.timeSinceStartup+1;Write("RUNNING");}
   else if(State=="running"&&!Running){RestoreClock();State="leave";Write("SUITE_FINISHED");EditorApplication.isPlaying=false;}
   else if(State=="running"&&EditorApplication.timeSinceStartup-started>300)throw new Exception("Harness watchdog 300s");
  }catch(Exception ex){Directory.CreateDirectory(Folder);File.WriteAllText(Path.Combine(Folder,"HarnessError.txt"),ex.ToString());RestoreClock();State="leave";Write("SUITE_FAILED");if(EditorApplication.isPlaying)EditorApplication.isPlaying=false;else{Index++;State="enter";}}
 }
